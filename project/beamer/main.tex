\begin{frame}{Agentes que reescriben su propio plan}
  \begin{columns}[T]
    \begin{column}{0.53\textwidth}
      Una clase con nombre propio: Los \textbf{agentes auto-evolutivos}
      \parencite{gao2026selfevolving}.
      \begin{itemize}
        \item \textbf{Voyager} programa y repara sus propias habilidades en
          Minecraft \parencite{wang2023voyager}.
        \item \textbf{AutoGPT} reescribe su lista de tareas; \textbf{Reflexion},
          su memoria tras fallar \parencite{shinn2023reflexion}.
        \item \textbf{The AI Scientist} editó su código para extender su propio
          límite de tiempo \parencite{lu2024aiscientist}.
      \end{itemize}
      \vspace{0.3em}
      {\scriptsize\color{mcmBlack!65}Instancia de calibración: Un motor de
      ejecución agéntica donde esa decisión se tomó por intuición.
      
      Aporta cuestionamiento, no evidencia.\par}
    \end{column}
    \begin{column}{0.43\textwidth}
      \begin{alertblock}{El problema}
        Se despliega sin un criterio medido de \emph{cuándo} ejercerla: la
        dimensión \emph{when} sigue abierta.
      \end{alertblock}
      \begin{exampleblock}{La pregunta}
        ¿Bajo qué condiciones del entorno un agente que reescribe su plan
        supera a uno con plan estático, y a partir de qué costo de
        reescritura la ventaja se invierte?
      \end{exampleblock}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Un problema clásico con otro vocabulario}
  \centering\footnotesize
  \renewcommand{\arraystretch}{1.18}
  \setlength{\tabcolsep}{4pt}
  % Secciones de Russell & Norvig (2020), solo como referencia privada:
  %   plan §11.4; monitoreo de ejecución y replanificación §11.5;
  %   observabilidad parcial §2.3; lazo abierto/cerrado §11.5.
  \begin{tabular}{@{}l l l@{}}
    \toprule
    \textbf{Motor agéntico} & \textbf{Russell \& Norvig (2020)} & \textbf{En el modelo} \\
    \midrule
    Programa editable           & Plan                                 & Líneas \texttt{GOTO}, \texttt{PICK}, \texttt{DELIVER} \\
    Revisarlo contra lo que ve  & Monitoreo de ejecución               & Divergencia creencia $\neq$ mundo \\
    Reescribirlo                & Replanificación                      & Oráculo: competencia $p$, costo $c$ \\
    Editar la ejecución siguiente & Lazo abierto, reinicio en frío     & \CR{} (suelta la carga) \\
    Editar la ejecución en curso  & Lazo cerrado, reparación en caliente & \CL{} \\
    Lo que tiene a la vista     & Observabilidad parcial               & Radio de visión $r$ \\
    Pensar mientras todo cambia & Metarazonamiento                     & \texttt{DELIBERAR} ocupa $c$ ticks \\
    \bottomrule
  \end{tabular}

  \vspace{0.9em}
  \begin{minipage}{0.86\linewidth}\centering\small
    El beneficio de replanificar crece con el dinamismo y su costo es casi
    constante: \textbf{debería existir un cruce}, y nadie lo había medido
    \parencite{zilberstein2011metareasoning,fox2006plan}.
  \end{minipage}
\end{frame}

\begin{frame}{El mundo: un recolector en una retícula que se mueve}
  \begin{columns}[c]
    \begin{column}{0.4\textwidth}
      \centering\figura{mundo.png}{0.8\textheight}
    \end{column}
    \begin{column}{0.58\textwidth}
      \small
      MESA 3.5.1 \parencite{kazil2020mesa}: retícula $15\times15$, 15\,\%
      de obstáculos, 8 recursos, capacidad 3, 400 ticks.
      \begin{itemize}
        \item $\boldsymbol\lambda$: probabilidad por tick de que un recurso
          derive a una celda vecina.
        \item $\boldsymbol c$: ticks que ocupa \texttt{DELIBERAR}; el mundo
          sigue moviéndose y el plan \textbf{nace desactualizado}.
        \item $\boldsymbol p$: competencia del oráculo; si falla, devuelve un
          plan \emph{plausible y equivocado}.
        \item $\boldsymbol r$: radio de visión; fuera de él, solo la creencia.
      \end{itemize}
      \vspace{0.2em}
      Utilidad neta $= 10 \times \text{entregas}$ \gris{($\kappa=0$: el único
      castigo por pensar es el tiempo).}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Ocho arquitecturas, un mismo cuerpo}

  \renewcommand{\arraystretch}{1.12}

  \begin{tabular}{@{}l l l l@{}}
    \toprule
    % R&N (2020): RX y MB §2.4; CL §11.5 (monitoreo en línea);
    % KB cap. 7 (fluentes, §7.7); BAY §14.2 (filtrado) y §16.1 (utilidad esperada).
    \textbf{Política} & & \textbf{Tipo} & \textbf{Cuándo replanifica} \\
    \midrule
    \multicolumn{4}{@{}l}{\gris{\emph{No planifican}}} \\
    \RX  & \gris{RX}  & Reflejo simple             & Nunca: va al recurso visible más cercano \\
    \MB  & \gris{MB}  & Reflejo con modelo         & Nunca: va al más cercano según su creencia \\
    \midrule
    \multicolumn{4}{@{}l}{\gris{\emph{Planifican con A* y un oráculo}}} \\
    \OL  & \gris{OL}  & Lazo abierto               & Solo cuando el plan falla \\
    \CR  & \gris{CR}  & Lazo abierto con reinicio  & Al ver la divergencia, en frío (suelta la carga) \\
    \CL  & \gris{CL}  & Lazo cerrado, en línea     & Al ver la divergencia, en caliente \\
    \GOV & \gris{GOV} & Lazo cerrado acotado       & Como \texttt{CL}, hasta agotar $B$ reparaciones \\
    \midrule
    \multicolumn{4}{@{}l}{\gris{\emph{Preguntan qué saben antes de pagar}}} \\
    \KB  & \gris{KB}  & Lógico, con fluentes       & Solo si deduce que el plan ya no se cumple \\
    \BY  & \gris{BAY} & Filtro + utilidad esperada & Solo si reparar cuesta menos en esperanza \\
    \bottomrule
  \end{tabular}

  \vspace{0.7em}
  \small Mismo cuerpo, mismo oráculo, misma regla de costo: lo único que cambia
  es \textbf{cuándo} se invoca al replanificador.
\end{frame}

\begin{frame}{Mismo replanificador, mismo precio}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{alertblock}{Antes (hasta el 2026-10-09)}
        \small \OL{} replanificaba \textbf{gratis} tras un fallo (A* exacto,
        0 ticks), mientras \CL{} pagaba $c$ ticks y la competencia $p$.
        Se comparaba una reparación pagada contra una gratuita.
      \end{alertblock}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{exampleblock}{Ahora}
        \small Los planes de rutina son gratis; \textbf{toda} reparación, la
        dispare un fallo o una divergencia, pasa por el mismo oráculo y cuesta
        $c$. El presupuesto $B$ acota solo lo proactivo.
      \end{exampleblock}
    \end{column}
  \end{columns}
  \vspace{0.6em}
  \small
  \begin{itemize}
    \item \textbf{No sobrevivieron:} la dominancia del lazo abierto (hasta 108
      puntos), la observabilidad como la variable que lo decide todo y el
      «rescate» por gobernanza.
    \item \textbf{Control exacto que la vigila:} \GOV{} con $B=0$ $\equiv$
      \OL{} corrida a corrida: 180 pares, 0 distintos.
  \end{itemize}
\end{frame}

\begin{frame}{Un agente basado en conocimiento}
  \begin{columns}[T]
    \begin{column}{0.46\textwidth}
      \footnotesize
      \textbf{TELL} de fluentes, $\mathit{Visto}(r,c,t)$ y $\neg\mathit{En}(r,c,t)$,
      más el axioma de deriva:
      \[
        \mathit{En}(r,c,t) \Rightarrow \textstyle\bigvee_{c'\in N^{+}(c)} \mathit{En}(r,c',t{+}1)
      \]
      Región posible: $R(r,t) = E^{\delta_t}\bigl(R(r,t{-}1)\bigr) \setminus V_t$.
    \end{column}
    \begin{column}{0.5\textwidth}
      \footnotesize
      \textbf{ASK} antes de pagar una reparación:
      \begin{enumerate}
        \item ¿Dónde puede estar $r$ ahora?
        \item ¿Qué recurso tengo asegurado? \gris{(el mejor peor caso)}
        \item ¿Sigue siendo ejecutable mi plan? Solo \emph{inválido} al
          \llegada{} dispara la reparación.
      \end{enumerate}
    \end{column}
  \end{columns}
  \vspace{0.1em}
  \centering\figura{tira-kb.png}{0.48\textheight}
\end{frame}

\begin{frame}{Un agente bayesiano: la creencia es correcta}
  \begin{columns}[c]
    \begin{column}{0.55\textwidth}
      \small
      \begin{itemize}
        \item \textbf{Aprende $\lambda$ mirando:} previa $\mathrm{Beta}(1,1)$,
          $\hat\lambda = (1+m)/(2+n)$. Error mediano 0.083, \textbf{0.020} y
          0.003 con $r=1$, 3 e $\infty$.
        % Filtrado: R&N (2020) §14.2; utilidad esperada: §16.1.
        \item \textbf{Filtra} cada recurso: el soporte coincide tick a
          tick con la región lógica $R(r,t)$.
        \item \textbf{Decide} por utilidad esperada: repara si
          \[
            c \;<\; (1-q)\,(c+k)
          \]
          \vspace{-1.3em}

          \gris{con $q$ = probabilidad de que el recurso siga ahí al llegar.}
      \end{itemize}
    \end{column}
    \begin{column}{0.43\textwidth}
      \centering\figura{fiabilidad.png}{0.66\textheight}

      \footnotesize Diagrama de fiabilidad: el filtro está calibrado (HB5).
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{…decide peor}

  \centering\figura{mecanismo.png}{0.37\textheight}

  \nota{$\lambda=0.5$, $r=3$, $c=5$: \texttt{CL} = \CL, \texttt{KB} = \KB,
  
  \texttt{BAY} = \BY{} (medias por corrida $\pm$ EE).}
  \vspace{0.3em}
  \raggedright\small
  \begin{itemize}
    \item \textbf{HB1 rechazada:} \BY{} $-$ \KB{} $=$ \dif{-16.0}{4.3}.

  \end{itemize}

\begin{frame}{Para llevarse}
  \begin{enumerate}
    \item Al mismo precio, reparar al ver la divergencia y esperar a que el
      plan falle \textbf{casi empatan}.
    \item La frontera real separa \textbf{deliberar de no deliberar}: con
      $\lambda$ alto y $c\ge2$ gana el agente que nunca delibera.
    \item \textbf{Reparar en caliente} vale desde $\lambda=0.1$ (+16 a +22): la
      conclusión más robusta.
    \item Saber qué se sabe ayuda solo si \textbf{evita pagar reparaciones};
      la probabilidad correcta no salva un mal modelo de costos.
    \item En un despliegue hay que medir \textbf{$\lambda$ y $c$}, no $p$; y
      $\lambda$ se puede medir.
  \end{enumerate}
  \vspace{0.4em}
  \begin{exampleblock}{En una frase}
    Decidir cuándo reescribir el plan es, antes que nada, decidir si deliberar.
  \end{exampleblock}
\end{frame}
